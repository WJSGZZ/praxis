\begin{theorem}[Nonnegative cumulative flow]
If $q(t)\geq 0$ for $0\leq t\leq t_f$, then $J\geq 0$.
\end{theorem}
\begin{proof}
The integral of a nonnegative function on the stated interval is nonnegative.
\end{proof}
\begin{algorithm}[htbp]
\caption{Compare feasible candidate policies}
\begin{algorithmic}[1]
\Require Candidate policies $P$, feasibility test $F$, objective $J$
\State $P_F\gets\{p\in P:F(p)\}$
\If{$P_F=\varnothing$}
  \State \Return No feasible policy
\EndIf
\State \Return A policy minimizing $J(p)$ over $P_F$
\end{algorithmic}
\end{algorithm}
